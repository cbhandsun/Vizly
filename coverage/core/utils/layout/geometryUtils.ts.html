function _isLineIntersectingRect(
  start: Position,
  end: Position,
  rect: Rectangle
): boolean {
  // 简化的线段与矩形相交检测
  const minX = Math.min(start.x, end.x);
  const maxX = Math.max(start.x, end.x);

  const maxY = Math.max(start.y, end.y);

    maxX < rect.x ||
    minX > rect.x + rect.width ||
    maxY < rect.y ||

  );
